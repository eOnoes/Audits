import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { join } from 'node:path';
import { execFileSync } from 'node:child_process';
import { fileURLToPath } from 'node:url';
const root = '[PRIVATE_PRODUCT_ROOT]';
const dir = join(root, 'docs/reports/v3-coordinator-20260917');
const sha = b => createHash('sha256').update(b).digest('hex');
const receiptBytes = readFileSync(join(dir, 'receipt.json'));
const r = JSON.parse(receiptBytes);
const indexMatches = [];
assert.equal(r.kind, 'producer-dormant-coordinator-synthetic-ports-not-physical-composition');
assert.equal(r.unchanged, true); assert.deepEqual(r.identitiesBefore, r.identitiesAfter);
for (const identity of r.identitiesBefore) {
  const b = readFileSync(join(root, identity.path)); assert.equal(b.length, identity.byteLength, identity.path);
  assert.equal(sha(b), identity.sha256, identity.path);
  if (!identity.path.startsWith('.test-dist/')) {
    const indexed = execFileSync('git', ['-c', 'safe.directory=[PRIVATE_REPOSITORY_ROOT]',
      'show', ':product/companion-v1/' + identity.path], { cwd: root, windowsHide: true, maxBuffer: 4_000_000 });
    const exact = sha(indexed) === identity.sha256;
    const configuration = ['tsconfig.json', 'tsconfig.test.json'].includes(identity.path);
    if (!exact) {
      assert(configuration, 'source byte mismatch: ' + identity.path);
      assert.equal(indexed.toString('utf8'), b.toString('utf8').replaceAll('\r\n', '\n'), 'only CRLF -> LF: ' + identity.path);
      assert.deepEqual(JSON.parse(indexed.toString('utf8')), JSON.parse(b.toString('utf8')));
    }
    indexMatches.push({ path: identity.path, executedSha256: identity.sha256, indexSha256: sha(indexed),
      relation: exact ? 'byte-identical' : 'executed-config-CRLF-index-LF-only' });
  }
}
assert.equal(sha(readFileSync(join(dir, 'verify.mjs'))), r.helperSha256);
assert.equal(sha(readFileSync(process.execPath)), r.nodeExecutableSha256);
assert.equal(r.runs.length, 2);
for (const run of r.runs) {
  const b = readFileSync(join(dir, run.rawTap.path));
  assert.equal(b.length, run.rawTap.byteLength); assert.equal(sha(b), run.rawTap.sha256);
  assert.equal(run.exitCode, 0); assert.equal(run.errorCode, null); assert.equal(run.signal, null);
  const text = b.toString('utf8'); assert.match(text, /^# fail 0$/m); assert.match(text, /^# cancelled 0$/m);
  assert.equal((text.match(/^ok \d+ - /gm) ?? []).length, run.name === 'focused' ? 136 : 2131);
  assert.match(text, new RegExp('^# tests ' + (run.name === 'focused' ? 136 : 2131) + '$', 'm'));
  assert.match(text, new RegExp('^# skipped ' + (run.name === 'focused' ? 0 : 2) + '$', 'm'));
  for (const diagnostic of ['53 synthetic paired appends', '32 before/after append seam abort cuts',
    '18 before/after bootstrap seam abort cuts', '16 settled append seam deadline cuts']) assert(text.includes(diagnostic));
}
const result = { result: 'verified', identities: r.identitiesBefore.length, indexMatches, runs: r.runs.map(x => ({ name: x.name,
  summary: x.summary, skipLines: x.skipLines })), receiptSha256: sha(receiptBytes) };
if (process.argv.includes('--capture')) {
  for (const path of ['tsconfig.json', 'tsconfig.test.json']) {
    writeFileSync(join(dir, 'EXECUTED_' + path + '.txt'), readFileSync(join(root, path)), { flag: 'wx' });
  }
  writeFileSync(join(dir, 'check-artifacts.mjs'), readFileSync(fileURLToPath(import.meta.url)), { flag: 'wx' });
  writeFileSync(join(dir, 'EXECUTION_INDEX_MATCH.json'), JSON.stringify(result, null, 2) + '\n', { flag: 'wx' });
}
console.log(JSON.stringify(result, null, 2));
